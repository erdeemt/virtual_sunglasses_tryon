// 1. Device & Browser Tespiti
function detectCapabilities() {
  const isIOS = /iPad|iPhone|iPod/.test(navigator.userAgent);
  const isAndroid = /Android/.test(navigator.userAgent);
  const isMobile = isIOS || isAndroid;
  const hasWebGL = !!document.createElement('canvas').getContext('webgl');
  const hasWebGPU = !!navigator.gpu; // Gelecek için
  const hasARKit = isIOS && window.ARKit; // iOS 12+ (opsiyonel)
  const hasARCore = isAndroid && navigator.xr; // WebXR desteği

  return {
    isIOS,
    isAndroid,
    isMobile,
    hasWebGL,
    hasWebGPU,
    hasARKit,
    hasARCore,
    recommendedMode: (hasARKit || hasARCore) ? 'native-ar' : 'mediapipe-web'
  };
}

Adım 1: Device Detection Ekle
javascript
// utils/deviceDetector.js
export function getDeviceProfile() {
  const ua = navigator.userAgent;
  const isIOS = /iPad|iPhone|iPod/.test(ua);
  const isAndroid = /Android/.test(ua);
  const isMobile = isIOS || isAndroid;
  
  // WebXR desteği (ARCore/ARKit için)
  const hasWebXR = !!navigator.xr;
  
  // WebGL desteği
  const hasWebGL = !!document.createElement('canvas').getContext('webgl');
  
  // Önerilen mod
  let mode = 'mediapipe'; // default
  if (hasWebXR && isIOS) mode = 'arkit';
  if (hasWebXR && isAndroid) mode = 'arcore';
  if (!hasWebGL) mode = 'fallback-2d';
  
  return { isIOS, isAndroid, isMobile, hasWebXR, hasWebGL, mode };
}
Adım 2: Dinamik Yükleme
javascript
// main.js
import { getDeviceProfile } from './utils/deviceDetector.js';

const profile = getDeviceProfile();

if (profile.mode === 'arkit' || profile.mode === 'arcore') {
  // WebXR + Native AR yükle
  import('./ar-webxr.js').then(module => module.init());
} else if (profile.mode === 'mediapipe') {
  // MediaPipe + Three.js yükle
  import('./mediapipe-three.js').then(module => module.init());
} else {
  // Fallback: 2D overlay
  import('./fallback-2d.js').then(module => module.init());
}

İlk Adım: Webcam + Face Mesh
MediaPipe Face Mesh’i kur, webcam feed’ini al, 468 landmark’ı ekrana çizdir.

Hedef: Yüz hareket ettikçe landmarkların real-time güncellendiğini görmek.

Örnek iskelet (index.html):

xml
<video id="input_video" autoplay playsinline></video>
<canvas id="output_canvas"></canvas>
<script type="module">
  import { FaceMesh } from '@mediapipe/face_mesh';
  import { Camera } from '@mediapipe/camera_utils';

  const video = document.getElementById('input_video');
  const canvas = document.getElementById('output_canvas');
  const ctx = canvas.getContext('2d');

  const faceMesh = new FaceMesh({
    locateFile: (file) => `https://cdn.jsdelivr.net/npm/@mediapipe/face_mesh/${file}`,
  });

  faceMesh.setOptions({
    maxNumFaces: 1,
    refineLandmarks: true,
    minDetectionConfidence: 0.5,
    minTrackingConfidence: 0.5,
  });

  faceMesh.onResults((results) => {
    ctx.clearRect(0, 0, canvas.width, canvas.height);
    ctx.drawImage(results.image, 0, 0, canvas.width, canvas.height);
    if (results.multiFaceLandmarks) {
      for (const landmarks of results.multiFaceLandmarks) {
        // 468 landmark'ı çiz (örnek: 168, 143, 2, 372)
        landmarks.forEach((point, i) => {
          const x = point.x * canvas.width;
          const y = point.y * canvas.height;
          ctx.beginPath();
          ctx.arc(x, y, 2, 0, 2 * Math.PI);
          ctx.fillStyle = 'red';
          ctx.fill();
        });
      }
    }
  });

  const camera = new Camera(video, {
    onFrame: async () => {
      await faceMesh.send({ image: video });
    },
    width: 640,
    height: 480,
  });
  camera.start();
</script>

javascript
// landmarks: [x, y, z] (0–1 arası normalize)
const leftEye = landmarks[143];
const rightEye = landmarks[372];
const noseTip = landmarks[2];
const forehead = landmarks[168];

// Gözler arası mesafe (pupillary distance proxy)
const eyeDistance = Math.sqrt(

  Math.pow((leftEye.y - rightEye.y) * canvas.height, 2)
);

// Gözlük pozisyonu (gözlerin ortası, burun köprüsü hizası)

const glassesY = ((leftEye.y + rightEye.y) / 2) * canvas.height;

CLAUDE 


1. Kamera katmanı — neden TrueDepth şart oldu

iPhone X ve sonrasındaki ön kamerada bir IR nokta projektörü var: yüzüne ~30.000 kızılötesi nokta atıyor, ayrı bir IR kamera bu noktaların desenini okuyor. Noktalar arasındaki bozulmadan structured light triangulation ile piksel başına derinlik çıkarıyorsun. Bu Face ID'yi çalıştıran donanımın aynısı, ve Warby Parker'ın feature'ı bu yüzden iPhone X ve üstüyle sınırlı.

Kritik nokta: bu derinlik haritası metrik. Yani milimetre cinsinden gerçek mesafe veriyor. Neden önemli olduğuna 3. maddede geleceğiz.

2. Yüz geometrisi ve poz

İki ayrı yol var, ikisini de bilmen lazım:

Monoküler yol (sadece RGB, derinlik yok):

Yüz tespiti — hafif bir CNN detektör, tipik olarak BlazeFace tarzı SSD mimarisi. Anchor tabanlı, tek geçişte bounding box + 6 kaba landmark. Mobil GPU'da ~1ms.
Landmark regresyonu — klasik yöntem 68 noktalı iBUG şeması (dlib'in ERT'si, cascade of regression trees). Modern yöntem MediaPipe Face Mesh: 468 noktalı yoğun mesh, direkt koordinat regresyonu yapan bir CNN.
Poz çözümü — elinde 2D landmark var, kanonik bir 3D yüz modelinin karşılık gelen noktaları var. solvePnP ile (EPnP + Levenberg-Marquardt refinement) rotasyon R ve translasyon t'yi çözüyorsun. Ama yüz landmarkları neredeyse düzlemsel olduğu için PnP burada kötü koşullanmış — pitch ve z-translasyon birbirine karışır, "flip" yapar.
Daha doğrusu 3D Morphable Model (3DMM) kullanmak. Blanz & Vetter'ın 1999'daki fikri: yüzü bir lineer bazda ifade et.

Snapchat için sorun değil — köpek kulağı yüzün oranında ölçeklenir, kimse fark etmez. Gözlük için felaket. Çünkü:

Gerçek gözlüğün fiziksel ölçüleri sabit (lens genişliği 49mm, köprü 21mm, sap 145mm — çerçevenin içine yazan o rakamlar)
Senin yüz genişliğin 130mm mi 148mm mi, bu çerçevenin yüzünde nasıl duracağını tamamen değiştirir
Yanlış ölçekte render edersen her çerçeve herkese "mükemmel" oturur, ki bu bir alışveriş aracı olarak feature'ı işe yaramaz hale getirir

Çözümler:

Metrik derinlik sensörü (TrueDepth / LiDAR) — doğrudan mm veriyor. En temiz yol, WP'nin yolu.
İstatistiksel önsel — yetişkin ortalama pupil arası mesafe (PD) ~63mm, std ~3.5mm. Bunu ölçek sabiti olarak kullanırsın. Ucuz ama %5-6 hata payıyla yaşarsın.
Bilinen boyutlu referans nesne — kredi kartını alnına tut (85.6mm, ISO/IEC 7810 standardı sağ olsun). Bazı web VTO çözümleri hâlâ bunu yapıyor.
Kamera intrinsics — odak uzaklığı ve autofocus mesafe metadata'sı, çoğu web ortamında güvenilir değil.

WP zaten TrueDepth'i önce PD ölçmek ve çerçeve önermek için kullanmıştı, virtual try-on ondan sonra geldi.

4. Yerleştirme çözücüsü — "unique placement"

Burası ticari sırrın olduğu yer. Naif yaklaşım: gözlüğü göz landmarklarının ortasına koy, kafa pozunu uygula, bitti. Sonuç sahte görünür, ve WP'nin ekibi de tam olarak bunu söylüyor — sanal nesnenin yüze "yaklaşık" oturması yetmiyor, kulak, göz ve burun yerleşimiyle birebir eşleşmesi gerekiyor.

Doğru yaklaşım fiziği taklit etmek. Gerçek gözlük üç noktada durur:

Burun sırtı (nazal dorsum) — burun pedleri ya da köprü buraya oturur
Sol kulak üstü (superior helix bağlantısı)
Sağ kulak üstü

Üç temas noktası rijit bir cismin pozunu belirler. Yani çözücü şuna benziyor:

1. Simetri düzlemi kısıtı: çerçevenin orta düzlemi yüzün sagital düzlemine hizalanır
   → yaw ve roll'u kilitler, 6DoF'u 3DoF'a indirir (pitch, y, z)

2. Burun teması: çerçevenin nose pad geometrisinden yüz mesh'ine ray-cast at,
   ilk kesişimi bul → çerçeveyi yüzey normali boyunca temasa kadar ilerlet

3. Kulak teması: çerçeveyi burun temas noktası etrafında döndür,
   sap uçları kulak landmarklarına değene kadar → pantoskopik açıyı belirler

4. Penetrasyon kontrolü: sapların yanak/şakak mesh'ini kesip kesmediğini kontrol et
   Kesiyorsa çerçeve dar demektir → ya sapları esnet ya "fit" skorunu düşür

Bu aslında temas kısıtlı küçük bir optimizasyon problemi. ICP benzeri birkaç iterasyonla ya da doğrudan analitik olarak çözülebilir. Çıktı sadece bir pozdan ibaret değil — aynı zamanda bir uyum skoru üretiyor, ki WP bunu ürün önerisinde kullanıyor.

Buradaki incelik: burun köprüsünün açısı (bazı insanlarda dik, bazılarında yayvan) gözlüğün ne kadar aşağı ineceğini belirler. Aynı çerçeve iki farklı yüzde farklı yükseklikte oturur. Sabit bir offset kullanırsan bu tamamen kaybolur, ve fake görünmesinin bir numaralı sebebi budur.

5. Render

Fotogerçekçilik burada kazanılıyor ya da kaybediliyor. WP'nin ekibi çerçeveleri piksel piksel doğru göstermek için 3D sanatçılarla defalarca revizyon yaptıklarını söylüyor.

PBR materyal. Metalik/pürüzlülük iş akışı, mikrofaset BRDF (GGX/Trowbridge-Reitz dağılımı + Smith geometri terimi). Titanyum çerçeve metallic=1, roughness~0.25. Asetat metallic=0 ama translucent — ışığı içinden geçirir, o yüzden ince bir subsurface yaklaşımı ya da en azından backlight transmission terimi lazım. WP'nin kendi vurguladığı şeylerden biri asetat çerçevede ışığın nasıl süzüldüğünü gösterebilmeleri.

Işık tahmini. Sanal nesnenin ışığı gerçek sahneyle uyuşmazsa beyin anında yakalıyor. ARKit'in yüz takibi modunda ARDirectionalLightEstimate veriyor: ambient yoğunluk (lümen), renk sıcaklığı (Kelvin), ve 2. dereceden küresel harmonik katsayıları — RGB başına 9 katsayı. Yüz, albedosu kabaca bilinen bir Lambert yüzeyi olduğu için aslında ters render ile bu SH'yi görüntüden de çıkarabilirsin.

Occlusion — en kritik tek numara. Sapların kafanın arkasına geçmesi lazım. Yüz mesh'ini renk yazma kapalı, derinlik yazma açık şekilde render ediyorsun (depth-only pass / occlusion mesh). Böylece z-buffer sapları doğru yerde kesiyor. Bunu yapmayan implementasyonlarda gözlük yüzün "üstünde yüzüyor" gibi durur.

Saç ayrı bir dert. Yüz mesh'i saçı içermiyor, o yüzden uzun saçlı birinde sap saçın önünden geçiyor. Çözüm: saç segmentasyon/matting ağı (U-Net tarzı) çalıştırıp alpha maskesiyle sapları maskelemek. Pahalı ve gerçekçilik farkı büyük.

Gölge. Tahmini ana ışık yönünden shadow map alıp yüz mesh'ine projekte ediyorsun, çarpımsal kompozit. En büyük gerçekçilik ipucu burun köprüsündeki temas gölgesi — o küçük koyu şerit olmadığı zaman gözlük yapışkan sticker gibi duruyor.

Lens kırılması. Camdan bakınca arkadaki şey kayar/büyür. Screen-space refraction: sahneyi bir texture'a render et, lens shader'ında kırılma vektörünü hesapla

7. Asset pipeline — kimsenin konuşmadığı kısım

Katalogdaki her SKU için:

Fotogrametri taraması ya da CAD modeli, sonra retopoloji (temiz quad mesh, ~5-15k üçgen)
UV unwrap + PBR texture set (albedo, roughness, metallic, normal)
Semantik anchor noktaları: köprü merkezi, burun pedi temas düzlemleri, menteşe eksenleri, sap ucu eğrisi — hepsi kanonik bir koordinat çerçevesinde. Yerleştirme çözücüsü bunlar olmadan generic çalışamaz.
Gerçek fiziksel ölçülerle kalibrasyon (o çerçevenin içine yazan 49□21-145 gibi rakamlar)
Mobil için LOD ve texture sıkıştırma (ASTC)

WP'nin 3D sanatçılarla "her çerçeve için pixel-perfect olana kadar" defalarca revizyon yaptığını söylemesi tam olarak bu. Yüzlerce SKU × bu emek = feature'ın gerçek maliyeti. Algoritma bir kez yazılıyor, asset pipeline sonsuza kadar sürüyor.

8. Web tarafı neden daha kötü

Sende derinlik sensörü yok, getUserMedia'dan gelen RGB var. Tipik stack:

MediaPipe Face Mesh veya TensorFlow.js, WASM+SIMD backend (WebGL backend'i küçük modellerde genelde daha yavaş, transfer overhead'i yüzünden)
468 landmark, "attention mesh" varyantı göz/dudak bölgesini yüksek çözünürlükte iyileştiriyor
three.js ile WebGL render, occlusion mesh yine depth-only pass
Kamera intrinsics'i bilmiyorsun → FOV'u ~60° varsayıp geçiyorsun
Ölçek → PD önseline dayalı tahmin

Sonuç: pozu iyi tutturursun, ölçeği tutturamazsın. WP web'de bir AR try-on denemişti ve kısa süre sonra kaldırdı; sonra native app'e ARKit ile geri döndüler. Site tarafındaki deneme daha çok 2D fotoğraf bazlı kompozit ("fotoğrafını yükle, üstüne bindirelim"), gerçek AR değil. Bu arada iOS app kullanıcılarının dörtte üçü virtual try-on kullanıyor — yani native'de tuttu.

Sektörde bunu servis olarak satan şirketler de var (Fittingbox, Perfect Corp, Ditto, Banuba, Jeeliz) — çoğu markanın kendi ekibi yerine bunlardan birini entegre ettiğini görürsün.

9. Sıfırdan yapacak olsan

MVP (bir hafta sonu): MediaPipe Face Mesh + three.js + tek bir GLB gözlük modeli. Landmark'lardan Procrustes ile rijit poz, PD=63mm sabit ölçek, yüz mesh'i depth-only occlusion. Çalışır, ama sahte durur.

Ciddi versiyon: ARKit/ARCore Augmented Faces ile metrik mesh → temas kısıtlı yerleştirme çözücüsü → SH ışık tahmini + shadow map → screen-space lens refraction → One Euro filtre → düzgün asset pipeline.

En çok kazanç getiren üç şey sırasıyla: occlusion (sapların kafanın arkasına geçmesi), temas gölgesi (burun köprüsündeki), ve doğru metrik ölçek. Bu üçü olmadan ne kadar iyi PBR yaparsan yap sticker gibi durur.
